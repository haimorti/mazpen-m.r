# קריינות מנוקדת

הטקסט שנשלח למנוע הדיבור. על המסך ובכתוביות נשאר הטקסט ללא ניקוד.

## scenes

1. שָׁלוֹם. בַּסִּרְטוֹן הַזֶּה נַכִּיר אֶת מַצְפֵּן זְכוּיּוֹת אֵיבָה, אֵזוֹר חָדָשׁ בָּאֲתָר הָאִישִׁי, שֶׁזָּמִין בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. מַצְפֵּן הַזְּכוּיּוֹת מְרַכֵּז בְּמָקוֹם אֶחָד אֶת כָּל הַזְּכוּיּוֹת וְהַהֲטָבוֹת שֶׁמַּגִּיעוֹת לָכֶם כְּנִפְגְּעֵי פְּעוּלוֹת אֵיבָה.

3. נִכְנָסִים אֵלָיו מֵהָאֵזוֹר הָאִישִׁי שֶׁלָּכֶם. בַּתַּפְרִיט בּוֹחֲרִים מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, אַחֵר כָּךְ כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז לוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל.

4. זֶה הַמָּסָךְ הָרָאשִׁי שֶׁל הַמַּצְפֵּן, וְהוּא בָּנוּי מֵאַרְבָּעָה אֵזוֹרִים.

5. בָּאֵזוֹר הַזֶּה בּוֹחֲרִים עֲבוּר מִי לְהַצִּיג אֶת הַהֲטָבוֹת, וְהוּא מוֹפִיעַ רַק אִם יֵשׁ לָכֶם יְלָדִים קְטִינִים.

6. בַּחִיפּוּשׂ אֶפְשָׁר לְחַפֵּשׂ הֲטָבָה לְפִי הַשֵּׁם שֶׁלָּהּ, אוֹ לְפִי נוֹשֵׂא — לְמָשָׁל דִּיּוּר, רֶכֶב אוֹ טִיפּוּל רְפוּאִי.

7. בָּאֵזוֹר הַהֲטָבוֹת שֶׁלִּי מוֹפִיעוֹת הַהֲטָבוֹת שֶׁכְּבָר בִּיקַּשְׁתֶּם, אוֹ שֶׁכְּבָר אוּשְּׁרוּ לָכֶם.

8. וּלְמַטָּה, בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת, מוֹפִיעוֹת כָּל הַהֲטָבוֹת שֶׁאַתֶּם עֲשׂוּיִים לִהְיוֹת זַכָּאִים לָהֶן.

9. זֶה הַמָּסָךְ הַמֶּרְכָּזִי, וְאֵלָיו נַחֲזוֹר בְּכָל פַּעַם. נַתְחִיל מֵהַהֲטָבוֹת שֶׁלִּי וְנִלְחַץ עַל אַחַת מֵהֶן.

10. בְּדַף הַהֲטָבָה מוֹפִיעִים לְמַעְלָה הַשֵּׁם וְהַקָּטֵגוֹרְיָה, מִתַּחְתָּם תֵּיבַת הַהֶסְבֵּר, וּמִתַּחְתֶּיהָ שׁוּרַת הַהֲטָבָה. שֵׁם תּוּכְלוּ לִרְאוֹת בְּכָל רֶגַע מַה הַסְּטָטוּס שֶׁלָּהּ.

11. כְּשֶׁהִגַּשְׁתֶּם בַּקָּשָׁה וַעֲדַיִין לֹא הִתְקַבְּלָה בָּהּ הַחְלָטָה, הַהֲטָבָה מוֹפִיעָה בִּסְטָטוּס בְּטִיפּוּל. הַסְּטָטוּס יִתְעַדְכֵּן כָּאן לְבַד.

12. הֲטָבָה שֶׁאוּשְּׁרָה, שֶׁתְּקוּפַת הַזַּכָּאוּת שֶׁלָּהּ עֲדַיִין בְּתוֹקֶף וְשֶׁנִּשְׁאֲרָה בָּהּ יִתְרָה, מוֹפִיעָה בִּסְטָטוּס פְּעִילָה. בְּהַטָּבָה כָּזֹאת אֶפְשָׁר לְהַגִּישׁ קַבָּלוֹת.

13. הַסְּטָטוּס הִסְתַּיְּימָה אוֹמֵר שֶׁהַהֲטָבָה נוּצְּלָה בִּמְלוֹאָהּ, אוֹ שֶׁתְּקוּפַת הַזַּכָּאוּת חָלְפָה. בְּהַטָּבָה כָּזֹאת כְּבָר אִי אֶפְשָׁר לְהַגִּישׁ קַבָּלוֹת.

14. אִם הַבַּקָּשָׁה נִדְחֲתָה, הַהֲטָבָה תּוֹפִיעַ בִּסְטָטוּס נִדְחֲתָה, וְסִיבַּת הַדְּחִייָה תּוֹפִיעַ עַל הַכַּרְטִיס עַצְמוֹ.

15. וְהַסְּטָטוּס פּוֹטֶנְצִיאָלִי מְסַמֵּן הֲטָבָה שֶׁעֲדַיִין לֹא בִּיקַּשְׁתֶּם, וְיִיתָּכֵן שֶׁאַתֶּם זַכָּאִים לָהּ. מִכָּאן אֶפְשָׁר לְהַגִּישׁ עָלֶיהָ בַּקָּשָׁה.

16. בְּהַטָּבָה פְּעִילָה, לְחִיצָה עַל "פְּרָטִים נוֹסָפִים" פּוֹתַחַת אֶת הַתְּמוּנָה הַמְּלֵאָה שֶׁל הַהֲטָבָה.

17. בְּמָסָךְ אֶחָד רוֹאִים אֶת כָּל מַה שֶׁקּוֹרֶה בְּהַטָּבָה אַחַת — הַפְּרָטִים שֶׁלָּהּ, כַּמָּה מִימַּשְׁתֶּם, וּמַה קָרָה לְכָל קַבָּלָה שֶׁהִגַּשְׁתֶּם.

18. לְמַעְלָה מוֹפִיעִים שֵׁם הַהֲטָבָה וְתֵיבַת הַהֶסְבֵּר, וּמִתַּחְתָּם פְּרָטֵי הַהֲטָבָה — הַסְּטָטוּס, תְּקוּפַת הַזַּכָּאוּת, הַסְּכוּם וְסוּג הַתַּשְׁלוּם.

19. בָּאֵזוֹר סִיכּוּם מִימּוּשׁ רוֹאִים כַּמָּה כְּבָר שׁוּלַּם לָכֶם, וְכַמָּה נוֹתַר לָכֶם לְמַמֵּשׁ מִתּוֹךְ סְכוּם הַהֲטָבָה.

20. וּמֵהַכַּפְתּוֹר הַכָּחוֹל אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה, יְשִׁירוֹת מִכָּאן.

21. וּבָאֵזוֹר בַּקָּשׁוֹת לְהֶחְזֵר מוֹפִיעָה כָּל קַבָּלָה שֶׁהִגַּשְׁתֶּם, עִם הַתַּאֲרִיךְ, הַסְּכוּם וּסְטָטוּס הַטִּיפּוּל בָּהּ. אֶפְשָׁר גַּם לִפְתּוֹחַ אֶת הַקַּבָּלָה עַצְמָהּ.

22. חוֹזְרִים לַמָּסָךְ הַמֶּרְכָּזִי וְיוֹרְדִים לַהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת שֶׁלִּי. אֵלֶּה הֲטָבוֹת שֶׁיִּיתָּכֵן שֶׁמַּגִּיעוֹת לָכֶם וַעֲדַיִין לֹא בִּיקַּשְׁתֶּם. לְכָל מְבוּטָּח יֵשׁ רְשִׁימָה אַחֶרֶת, לְפִי הַנְּתוּנִים הָאִישִׁיִּים שֶׁלּוֹ. נִלְחַץ עַל אַחַת מֵהֶן.

23. כָּל הֲטָבָה נִפְתַּחַת בְּאוֹתוֹ מִבְנֶה. לְמַעְלָה שֵׁם הַהֲטָבָה וְהַקָּטֵגוֹרְיָה, אַחֵר כָּךְ תֵּיבַת הֶסְבֵּר שֶׁאוֹמֶרֶת מַה הַהֲטָבָה נוֹתֶנֶת, כַּמָּה וּבְאֵיזֶה תְּנַאי, וּלְמַטָּה שׁוּרַת הַהֲטָבָה עִם הַסְּטָטוּס, מַה צָרִיךְ לַעֲשׂוֹת, וְכַפְתּוֹר הַפְּעוּלָה.

24. בְּכָל הֲטָבָה מוֹפִיעַ אֶחָד מִשְּׁנֵי הַכַּפְתּוֹרִים, וְכָל אֶחָד פּוֹתֵחַ מָסָךְ אַחֵר. בַּכַּרְטִיס כָּתוּב "לִבְדִיקַת זַכָּאוּת חֲדָשָׁה".

25. נִפְתָּח טוֹפֶס בַּקָּשָׁה בֶּן שְׁלוֹשָׁה שְׁלָבִים.

26. בַּכַּרְטִיס כָּתוּב "אֶפְשָׁר לִפְנוֹת לְעוֹבֵד הַשִּׁיקּוּם".

27. נִפְתָּח טוֹפֶס פְּנִייָה לְעוֹבֵד הַשִּׁיקּוּם.

28. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר, וְהַבַּקָּשָׁה נִכְנֶסֶת לְמַצְפֵּן.

29. הַבַּקָּשָׁה מוֹפִיעָה בְּדַף הַהֲטָבָה בִּסְטָטוּס "בְּטִיפּוּל", וּמִתְעַדְכֶּנֶת שָׁם לְבַד. אֵין צוֹרֶךְ לַעֲשׂוֹת דָּבָר.

30. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת, וּבְהֶמְשֵׁךְ הוּא יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.

## scenes-invoice

1. בַּסִּרְטוֹן הַזֶּה נִרְאֶה אֵיךְ מַגִּישִׁים חֶשְׁבּוֹנִית אוֹ קַבָּלָה לְהֶחְזֵר, דֶּרֶךְ מַצְפֵּן זְכוּיּוֹת אֵיבָה. הַמַּצְפֵּן זָמִין בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. הַגָּשָׁה דֶּרֶךְ הַמַּצְפֵּן מַגִּיעָה יָשָׁר לַתִּיק שֶׁלָּכֶם, וְאֶפְשָׁר לַעֲקוֹב אַחֲרֶיהָ בְּכָל רֶגַע.

3. נִכְנָסִים מֵהָאֵזוֹר הָאִישִׁי. בַּתַּפְרִיט בּוֹחֲרִים מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, אַחֵר כָּךְ כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז לוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל.

4. יֵשׁ שְׁתֵּי דְּרָכִים. אִם הַהֲטָבָה נִמְצֵאת בַּהֲטָבוֹת שֶׁלִּי, מַתְחִילִים מִשָּׁם. אִם לֹא, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת. בִּשְׁנֵי הַמִּקְרִים לוֹחֲצִים עַל הַהֲטָבָה שֶׁעֲבוּרָהּ יֵשׁ לָכֶם חֶשְׁבּוֹנִית.

5. בְּדַף הַהֲטָבָה לוֹחֲצִים עַל "פְּרָטִים נוֹסָפִים".

6. וְלוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל, לְהַגָּשַׁת חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה. מִכָּאן מַגִּישִׁים.

7. אִם הַהֲטָבָה לֹא מוֹפִיעָה בַּהֲטָבוֹת שֶׁלִּי, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת וְלוֹחֲצִים עָלֶיהָ. שִׂימוּ לֵב, לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית בַּמַּצְפֵּן. בָּעַמּוּד שֶׁנִּפְתָּח תֵּדְעוּ אֶת זֶה לְפִי הַכַּפְתּוֹר.

8. בָּעַמּוּד שֶׁנִּפְתָּח מוֹפִיעַ אֶחָד מִשְּׁנֵי הַכַּפְתּוֹרִים, וְכָל אֶחָד מֵהֶם עוֹשֶׂה דָּבָר אַחֵר. הַכַּפְתּוֹר הָרִאשׁוֹן מַגִּישׁ חֶשְׁבּוֹנִית. אִם הוּא מוֹפִיעַ, אֶפְשָׁר לְהַגִּישׁ כָּאן.

9. נִפְתָּח טוֹפֶס הַהַגָּשָׁה, בֶּן שְׁלוֹשָׁה שְׁלָבִים.

10. הַכַּפְתּוֹר הַשֵּׁנִי אֵינוֹ מַגִּישׁ חֶשְׁבּוֹנִית, אֶלָּא פּוֹנֶה לְעוֹבֵד הַשִּׁיקּוּם.

11. נִפְתָּח טוֹפֶס פְּנִייָה, לֹא טוֹפֶס הַגָּשָׁה.

12. מִשְּׁתֵּי הַדְּרָכִים מַגִּיעִים לְאוֹתוֹ טוֹפֶס בֶּן שְׁלוֹשָׁה שְׁלָבִים. בַּשָּׁלָב הָרִאשׁוֹן בּוֹדְקִים אֶת פְּרָטֵי הַהֲטָבָה וְלוֹחֲצִים "הַבָּא".

13. בַּשָּׁלָב הַשֵּׁנִי מְצָרְפִים אֶת חֶשְׁבּוֹנִית הַמַּס אוֹ הַקַּבָּלָה. גּוֹרְרִים אֶת הַקּוֹבֶץ לַמִּסְגֶּרֶת, אוֹ לוֹחֲצִים "בְּחַר קוֹבֶץ".

14. אַחֵר כָּךְ מְמַלְּאִים אֶת פְּרָטֵי הַחֶשְׁבּוֹנִית — מִסְפָּר, תַּאֲרִיךְ וּסְכוּם כּוֹלֵל מַס עֵרֶךְ מוּסָף. תְּקוּפָה וְכַמּוּת מְמַלְּאִים לְפִי הַצּוֹרֶךְ.

15. הַכַּפְתּוֹר "הוֹסֵף חֶשְׁבּוֹנִית" מְאַפְשֵׁר לְהוֹסִיף עוֹד חֶשְׁבּוֹנִית לְאוֹתָהּ בַּקָּשָׁה. כְּשֶׁסִּייַמְתֶּם, לוֹחֲצִים "הַבָּא".

16. קוֹרְאִים אֶת הַהַצְהָרָה וּמְסַמְּנִים אֶת הַתֵּיבָה.

17. חוֹתְמִים בַּמִּסְגֶּרֶת — בָּעַכְבָּר בַּמַּחְשֵׁב, אוֹ בָּאֶצְבַּע בַּטֶּלֶפוֹן. וְאָז שׁוֹלְחִים.

18. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר: תּוֹדָה, הַפְּנִייָה נִקְלְטָה לְהֶמְשֵׁךְ טִיפּוּל.

19. הַגָּשַׁת בַּקָּשָׁה אֵינָהּ אִישּׁוּר אוֹטוֹמָטִי. הַזַּכָּאוּת תִּיבָּדֵק לְפִי הַקְּרִיטֶרְיוֹנִים שֶׁנִּקְבְּעוּ בַּחוֹק.

20. הַבַּקָּשָׁה תּוֹפִיעַ בְּדַף הַהֲטָבָה תַּחַת "בַּקָּשׁוֹת לְהֶחְזֵר", עִם הַסְּכוּם וּסְטָטוּס הַטִּיפּוּל בָּהּ. שָׁם תֵּדְעוּ אִם הַקַּבָּלָה שׁוּלְּמָה.

21. וּלְבַסּוֹף, לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ דֶּרֶךְ הַמַּצְפֵּן. כְּשֶׁלֹּא מוֹפִיעַ הַכַּפְתּוֹר הַגָּשַׁת בַּקָּשָׁה, הַהַגָּשָׁה נַעֲשֵׂית בְּאֶמְצָעוּת הַעֲלָאַת מִסְמָכִים בָּאֵזוֹר הָאִישִׁי. שֵׁם בּוֹחֲרִים נוֹשֵׂא שִׁיקּוּם, קָטֵגוֹרְיָה פְּנִיּוֹת וּמִסְמָךְ פְּנִייָה, מְצָרְפִים אֶת הַקּוֹבֶץ וְשׁוֹלְחִים.

22. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת, וּבְהֶמְשֵׁךְ הוּא יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.
