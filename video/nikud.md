# קריינות מנוקדת

הטקסט שנשלח למנוע הדיבור. על המסך ובכתוביות נשאר הטקסט ללא ניקוד.

## scenes

1. שָׁלוֹם. נַכִּיר אֶת מַצְפֵּן זְכוּיּוֹת אֵיבָה, אֵזוֹר חָדָשׁ בָּאֲתָר הָאִישִׁי, שֶׁפָּתוּחַ בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. מַצְפֵּן הַזְּכוּיּוֹת מְרַכֵּז עֲבוּרְךָ, בְּמָקוֹם אֶחָד, אֶת הַזְּכוּיּוֹת וְהַהֲטָבוֹת שֶׁמַּגִּיעוֹת לְךָ כְּנִפְגַּע פְּעוּלוֹת אֵיבָה.

3. הַכְּנִיסָה הִיא מֵהָאֵזוֹר הָאִישִׁי שֶׁלְּךָ: מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז הַכַּפְתּוֹר הַכָּחוֹל.

4. זֶה הַמָּסָךְ הָרָאשִׁי. הוּא בָּנוּי מֵאַרְבָּעָה אֵזוֹרִים.

5. הַצָּגַת הֲטָבוֹת עֲבוּר: בּוֹחֲרִים עֲבוּר מִי לְהַצִּיג. מוֹפִיעַ רַק אִם יֵשׁ לְךָ יְלָדִים קְטִינִים.

6. חִיפּוּשׂ: לְפִי שֵׁם הַהֲטָבָה, אוֹ לְפִי נוֹשֵׂא, לְמָשָׁל דִּיּוּר, רֶכֶב אוֹ טִיפּוּל רְפוּאִי.

7. הַהֲטָבוֹת שֶׁלִּי: אֵלֶּה הַהֲטָבוֹת שֶׁכְּבָר בִּיקַּשְׁתָּ אוֹ שֶׁכְּבָר אוּשְּׁרוּ לְךָ.

8. וְהַהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת שֶׁלְּךָ: כָּל הַהֲטָבוֹת לָהֶן אַתָּה עָשׂוּי לִהְיוֹת זַכַּאי.

9. זֶה הַמָּסָךְ הַמֶּרְכָּזִי, וְאֵלָיו נַחֲזוֹר בְּכָל פַּעַם. נַתְחִיל מֵהַהֲטָבוֹת שֶׁלִּי, וְנִלְחַץ עַל אַחַת מֵהֶן.

10. בְּדַף הַהֲטָבָה: הַשֵּׁם וְהַקָּטֵגוֹרְיָה לְמַעְלָה, תֵּיבַת הַהֶסְבֵּר, וּמִתַּחְתָּן שׁוּרַת הַהֲטָבָה. תּוּכְלוּ לִרְאוֹת בְּכָל רֶגַע אֶת סְטָטוּס הַהֲטָבָה שֶׁלָּכֶם.

11. בְּטִיפּוּל: הַגָּשַׁת בַּקָּשָׁה וַעֲדַיִין לֹא הִתְקַבְּלָה בָּהּ הַחְלָטָה. הַסְּטָטוּס יִתְעַדְכֵּן כָּאן.

12. פְּעִילָה: הַהֲטָבָה אוּשְּׁרָה, תְּקוּפַת הַזַּכָּאוּת לֹא חָלְפָה, וְיֵשׁ בָּהּ יִתְרָה. אֶפְשָׁר לְהַגִּישׁ קַבָּלוֹת.

13. הִסְתַּיְּימָה: הַהֲטָבָה נוּצְּלָה בִּמְלוֹאָהּ, אוֹ שֶׁתְּקוּפַת הַזַּכָּאוּת חָלְפָה. לֹא נִיתָּן לְהַגִּישׁ בָּהּ קַבָּלוֹת.

14. נִדְחֲתָה: הַבַּקָּשָׁה נִדְחֲתָה, וְסִיבַּת הַדְּחִייָה מוֹפִיעָה עַל הַכַּרְטִיס.

15. פּוֹטֶנְצִיאָלִי: הֲטָבָה שֶׁעֲדַיִין לֹא בִּיקַּשְׁתָּ וְיִיתָּכֵן שֶׁאַתָּה זַכַּאי לָהּ. אֶפְשָׁר לְהַגִּישׁ בַּקָּשָׁה.

16. בְּהַטָּבָה פְּעִילָה, "פְּרָטִים נוֹסָפִים" פּוֹתֵחַ אֶת הַתְּמוּנָה הַמְּלֵאָה שֶׁל הַהֲטָבָה.

17. זֶה כָּל מַה שֶׁקּוֹרֶה בְּהַטָּבָה אַחַת, בְּמָסָךְ אֶחָד: הַפְּרָטִים, כַּמָּה מִימַּשְׁתָּ, וּמַה קָרָה לְכָל קַבָּלָה שֶׁהִגַּשְׁתָּ.

18. לְמַעְלָה: שֵׁם הַהֲטָבָה וְתֵיבַת הַהֶסְבֵּר, וּמִתַּחְתָּן פְּרָטֵי הַהֲטָבָה – הַסְּטָטוּס, תְּקוּפַת הַזַּכָּאוּת, הַסְּכוּם וְסוּג הַתַּשְׁלוּם.

19. סִיכּוּם מִימּוּשׁ: כַּמָּה כְּבָר שׁוּלַּם לְךָ, וְכַמָּה נוֹתַר לְךָ לְמַמֵּשׁ מִתּוֹךְ סְכוּם הַהֲטָבָה.

20. לְהַגָּשַׁת חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה – הַכַּפְתּוֹר הַכָּחוֹל, יְשִׁירוֹת מִכָּאן.

21. בַּקָּשׁוֹת לְהֶחְזֵר: כָּל קַבָּלָה שֶׁהִגַּשְׁתָּ – הַתַּאֲרִיךְ, הַסְּכוּם, סְטָטוּס הַטִּיפּוּל בָּהּ, וְאֶפְשָׁרוּת לִפְתּוֹחַ אֶת הַקַּבָּלָה עַצְמָהּ.

22. חוֹזְרִים לַמָּסָךְ הַמֶּרְכָּזִי, וְיוֹרְדִים לַהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת שֶׁלִּי. אֵלֶּה הֲטָבוֹת שֶׁיִּיתָּכֵן שֶׁמַּגִּיעוֹת לְךָ וַעֲדַיִין לֹא בִּיקַּשְׁתָּ. לְכָל מְבוּטָּח רְשִׁימָה אַחֶרֶת, לְפִי הַנְּתוּנִים הָאִישִׁיִּים שֶׁלּוֹ. לוֹחֲצִים עַל הֲטָבָה.

23. כָּל הֲטָבָה נִפְתַּחַת בְּאוֹתוֹ מִבְנֶה: שֵׁם הַהֲטָבָה וְהַקָּטֵגוֹרְיָה, תֵּיבַת הֶסְבֵּר שֶׁאוֹמֶרֶת מַה הַהֲטָבָה נוֹתֶנֶת, כַּמָּה וּבְאֵיזֶה תְּנַאי, וּלְמַטָּה שׁוּרַת הַהֲטָבָה עִם הַסְּטָטוּס, מַה צָרִיךְ לַעֲשׂוֹת, וְכַפְתּוֹר הַפְּעוּלָה.

24. בְּכָל הֲטָבָה מוֹפִיעַ אֶחָד מִשְּׁנֵי הַכַּפְתּוֹרִים, וְכָל אֶחָד פּוֹתֵחַ מָסָךְ אַחֵר: בַּכַּרְטִיס כָּתוּב "לִבְדִיקַת זַכָּאוּת חֲדָשָׁה".

25. נִפְתָּח טוֹפֶס בַּקָּשָׁה, שְׁלוֹשָׁה שְׁלָבִים.

26. בַּכַּרְטִיס כָּתוּב "אֶפְשָׁר לִפְנוֹת לְעוֹבֵד הַשִּׁיקּוּם".

27. נִפְתָּח טוֹפֶס פְּנִייָה לְעוֹבֵד הַשִּׁיקּוּם.

28. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר, וְהַבַּקָּשָׁה מוֹפִיעָה בַּמַּצְפֵּן.

29. הַבַּקָּשָׁה מוֹפִיעָה בְּדַף הַהֲטָבָה בִּסְטָטוּס "בְּטִיפּוּל", וּמִתְעַדְכֶּנֶת שָׁם לְבַד. אֵין צוֹרֶךְ לַעֲשׂוֹת דָּבָר.

30. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת. בְּהֶמְשֵׁךְ יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.

## scenes-invoice

1. אֵיךְ מַגִּישִׁים חֶשְׁבּוֹנִית אוֹ קַבָּלָה לְהֶחְזֵר, דֶּרֶךְ מַצְפֵּן זְכוּיּוֹת אֵיבָה. הַמַּצְפֵּן פָּתוּחַ בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. הַגָּשָׁה דֶּרֶךְ הַמַּצְפֵּן מַגִּיעָה יָשָׁר לַתִּיק שֶׁלְּךָ, וְאֶפְשָׁר לַעֲקוֹב אַחֲרֶיהָ בְּכָל רֶגַע.

3. מֵהָאֵזוֹר הָאִישִׁי: מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז הַכַּפְתּוֹר הַכָּחוֹל.

4. יֵשׁ שְׁתֵּי דְּרָכִים. אִם הַהֲטָבָה נִמְצֵאת בַּהֲטָבוֹת שֶׁלִּי, זֶה הַמָּקוֹם הַנָּכוֹן. אִם לֹא, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת. לוֹחֲצִים עַל הַהֲטָבָה שֶׁעֲבוּרָהּ יֵשׁ לְךָ חֶשְׁבּוֹנִית.

5. בְּדַף הַהֲטָבָה לוֹחֲצִים עַל "פְּרָטִים נוֹסָפִים".

6. וְלוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל: לְהַגָּשַׁת חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה. מִכָּאן מַגִּישִׁים.

7. אִם הַהֲטָבָה לֹא מוֹפִיעָה בַּהֲטָבוֹת שֶׁלִּי, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת וְלוֹחֲצִים עָלֶיהָ. שִׂימוּ לֵב: לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית בַּמַּצְפֵּן. בָּעַמּוּד שֶׁנִּפְתָּח תִּרְאוּ לְפִי הַכַּפְתּוֹר.

8. בָּעַמּוּד שֶׁנִּפְתָּח מוֹפִיעַ אֶחָד מִשְּׁנֵי הַכַּפְתּוֹרִים. רַק אֶחָד מֵהֶם מַגִּישׁ חֶשְׁבּוֹנִית: מַגִּישׁ חֶשְׁבּוֹנִית. אִם הוּא מוֹפִיעַ — אֶפְשָׁר לְהַגִּישׁ כָּאן.

9. נִפְתָּח טוֹפֶס הַהַגָּשָׁה, שְׁלוֹשָׁה שְׁלָבִים.

10. לֹא מַגִּישׁ חֶשְׁבּוֹנִית. פּוֹנֶה לְעוֹבֵד הַשִּׁיקּוּם.

11. נִפְתָּח טוֹפֶס פְּנִייָה, לֹא טוֹפֶס הַגָּשָׁה.

12. מִשְּׁתֵּי הַדְּרָכִים מַגִּיעִים לְאוֹתוֹ טוֹפֶס בֶּן שְׁלוֹשָׁה שְׁלָבִים. בַּשָּׁלָב הָרִאשׁוֹן בּוֹדְקִים אֶת פְּרָטֵי הַהֲטָבָה וְלוֹחֲצִים "הַבָּא".

13. בַּשָּׁלָב הַשֵּׁנִי מְצָרְפִים אֶת חֶשְׁבּוֹנִית הַמַּס אוֹ הַקַּבָּלָה: גּוֹרְרִים אֶת הַקּוֹבֶץ, אוֹ לוֹחֲצִים "בְּחַר קוֹבֶץ".

14. מְמַלְּאִים לְפִי הַחֶשְׁבּוֹנִית: מִסְפָּר, תַּאֲרִיךְ, סְכוּם כּוֹלֵל מַס עֵרֶךְ מוּסָף. תְּקוּפָה וְכַמּוּת — לְפִי הַצּוֹרֶךְ.

15. "הוֹסֵף חֶשְׁבּוֹנִית" מוֹסִיף עוֹד חֶשְׁבּוֹנִית לְאוֹתָהּ בַּקָּשָׁה. כְּשֶׁסִּייַמְתֶּם — "הַבָּא".

16. קוֹרְאִים אֶת הַהַצְהָרָה וּמְסַמְּנִים אֶת הַתֵּיבָה.

17. חוֹתְמִים בַּמִּסְגֶּרֶת — בָּעַכְבָּר בַּמַּחְשֵׁב, אוֹ בָּאֶצְבַּע בַּטֶּלֶפוֹן. וְאָז שׁוֹלְחִים.

18. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר: תּוֹדָה, פְּנִייָתְךָ נִקְלְטָה לְהֶמְשֵׁךְ טִיפּוּל.

19. הַגָּשַׁת בַּקָּשָׁה אֵינָהּ אִישּׁוּר אוֹטוֹמָטִי. הַזַּכָּאוּת תִּיבָּדֵק לְפִי הַקְּרִיטֶרְיוֹנִים שֶׁנִּקְבְּעוּ בַּחוֹק.

20. הַבַּקָּשָׁה תּוֹפִיעַ בְּדַף הַהֲטָבָה תַּחַת "בַּקָּשׁוֹת לְהֶחְזֵר", עִם הַסְּכוּם וּסְטָטוּס הַטִּיפּוּל בָּהּ. שָׁם תֵּדְעוּ אִם הַקַּבָּלָה שׁוּלְּמָה.

21. וּלְבַסּוֹף: לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ דֶּרֶךְ הַמַּצְפֵּן. כְּשֶׁלֹּא מוֹפִיעַ "הַגָּשַׁת בַּקָּשָׁה", הַהַגָּשָׁה נַעֲשֵׂית בְּאֶמְצָעוּת הַעֲלָאַת מִסְמָכִים בָּאֵזוֹר הָאִישִׁי. בּוֹחֲרִים נוֹשֵׂא שִׁיקּוּם, קָטֵגוֹרְיָה פְּנִיּוֹת, מִסְמָךְ פְּנִייָה, מְצָרְפִים אֶת הַקּוֹבֶץ, וְשׁוֹלְחִים.

22. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת. בְּהֶמְשֵׁךְ יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.
