# קריינות מנוקדת

הטקסט שנשלח למנוע הדיבור. על המסך ובכתוביות נשאר הטקסט ללא ניקוד.

## scenes

1. שָׁלוֹם. בַּסִּרְטוֹן הַזֶּה נַכִּיר אֶת מַצְפֵּן זְכוּיּוֹת אֵיבָה, אֵזוֹר חָדָשׁ בָּאֲתָר הָאִישִׁי, שֶׁזָּמִין בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. מַצְפֵּן הַזְּכוּיּוֹת מְרַכֵּז בְּמָקוֹם אֶחָד אֶת כָּל הַזְּכוּיּוֹת וְהַהֲטָבוֹת שֶׁמַּגִּיעוֹת לָכֶם כְּנִפְגְּעֵי פְּעוּלוֹת אֵיבָה.

3. נִכְנָסִים אֵלָיו מֵהָאֵזוֹר הָאִישִׁי שֶׁלָּכֶם. בַּתַּפְרִיט בּוֹחֲרִים מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, אַחַר כָּךְ כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז לוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל.

4. זֶה הַמָּסָךְ הָרָאשִׁי שֶׁל הַמַּצְפֵּן, וְהוּא בָּנוּי מֵאַרְבָּעָה אֵזוֹרִים.

5. בָּאֵזוֹר הַזֶּה בּוֹחֲרִים עֲבוּר מִי לְהַצִּיג אֶת הַהֲטָבוֹת, וְהוּא מוֹפִיעַ רַק אִם יֵשׁ לָכֶם יְלָדִים קְטִינִים.

6. בַּחִיפּוּשׂ אֶפְשָׁר לְחַפֵּשׂ הֲטָבָה לְפִי הַשֵּׁם שֶׁלָּהּ, אוֹ לְפִי נוֹסֵא — לְמָשָׁל דִּיּוּר, רֶכֶב אוֹ טִיפּוּל רְפוּאִי.

7. בָּאֵזוֹר הַהֲטָבוֹת שֶׁלִּי מוֹפִיעוֹת הַהֲטָבוֹת שֶׁכְּבָר בִּיקַּשְׁתֶּם, אוֹ שֶׁכְּבָר אוּשְּׁרוּ לָכֶם.

8. וּלְמַטָּה, בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת, מוֹפִיעוֹת כָּל הַהֲטָבוֹת שֶׁאַתֶּם עֲשׂוּיִים לִהְיוֹת זַכָּאִים לָהֶן.

9. נַחֲזוֹר לַמָּסָךְ הַמֶּרְכָּזִי, נַתְחִיל מֵהַהֲטָבוֹת שֶׁלִּי וְנִלְחַץ עַל אַחַת מֵהֶן.

10. בְּדַף הַהֲטָבָה מוֹפִיעִים לְמַעְלָה שֵׁם הַהֲטָבָה וְהַקָּטֵגוֹרְיָה שֶׁלָּהּ, וּמִתַּחְתָּם תֵּיבַת הֶסְבֵּר קְצָרָה הַמְּתָאֶרֶת אֶת מַהוּת הַהֲטָבָה.

11. בָּאֵזוֹר הֲטָבוֹת שֶׁלִּי מוֹפִיעוֹת הַהֲטָבוֹת מֵהַנּוֹסֵא הַזֶּה שֶׁכְּבָר בִּיקַּשְׁתֶּם אוֹ שֶׁכְּבָר אוּשְּׁרוּ לָכֶם, וּלְצַד כָּל אַחַת מֵהֶן תְּקוּפַת הַזַּכָּאוּת, סְכוּם הַהֲטָבָה, הַכַּמּוּת וְסוּג הַתַּשְׁלוּם.

12. לְצַד כָּל הֲטָבָה מוֹפִיעַ הַסְּטָטוּס שֶׁלָּהּ.

13. לְחִיצָה עַל "פְּרָטִים נוֹסָפִים" פּוֹתַחַת אֶת הַתְּמוּנָה הַמְּלֵאָה שֶׁל הַהֲטָבָה.

14. בְּמָסָךְ אֶחָד רוֹאִים אֶת כָּל מַה שֶׁקּוֹרֶה בְּהַטָּבָה אַחַת — הַפְּרָטִים שֶׁלָּהּ, כַּמָּה מִימַּשְׁתֶּם, וּמַה קָרָה לְכָל קַבָּלָה שֶׁהִגַּשְׁתֶּם.

15. לְמַעְלָה מוֹפִיעִים שֵׁם הַהֲטָבָה וְתֵיבַת הַהֶסְבֵּר, וּמִתַּחְתָּם פְּרָטֵי הַהֲטָבָה — הַסְּטָטוּס, תְּקוּפַת הַזַּכָּאוּת, הַסְּכוּם וְסוּג הַתַּשְׁלוּם.

16. בָּאֵזוֹר סִיכּוּם מִימּוּשׁ רוֹאִים כַּמָּה כְּבָר שׁוּלַּם לָכֶם, וְכַמָּה נוֹתַר לָכֶם לְמַמֵּשׁ מִתּוֹךְ סְכוּם הַהֲטָבָה.

17. וּמֵהַכַּפְתּוֹר הַכָּחוֹל אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה, יְשִׁירוֹת מִכָּאן.

18. וּלְמַטָּה, בְּבַקָּשׁוֹת לְהֶחְזֵר, מוֹפִיעָה כָּל קַבָּלָה שֶׁהִגַּשְׁתֶּם — מָתַי הִגַּשְׁתֶּם אוֹתָהּ, עַל אֵיזֶה סְכוּם, וּמַה קָרָה אִיתָּהּ. מִכָּאן אֶפְשָׁר גַּם לִפְתּוֹחַ אֶת הַקַּבָּלָה עַצְמָהּ.

19. חוֹזְרִים לַמָּסָךְ הַמֶּרְכָּזִי וְיוֹרְדִים לַהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת שֶׁלִּי. אֵלֶּה הֲטָבוֹת שֶׁיִּיתָּכֵן שֶׁמַּגִּיעוֹת לָכֶם וַעֲדַיִין לֹא בִּיקַּשְׁתֶּם. לְכָל מְבוּטָּח יֵשׁ רְשִׁימָה אַחֶרֶת, לְפִי הַנְּתוּנִים הָאִישִׁיִּים שֶׁלּוֹ. נִלְחַץ עַל אַחַת מֵהֶן.

20. כָּל הֲטָבָה נִפְתַּחַת בְּאוֹתוֹ מִבְנֶה. לְמַעְלָה שֵׁם הַהֲטָבָה, אַחַר כָּךְ תֵּיבַת הַהֶסְבֵּר, וּלְמַטָּה כַּפְתּוֹר הַפְּעוּלָה.

21. בְּכָל הֲטָבָה מוֹפִיעַ אֶחָד מִשְּׁנֵי כַּפְתּוֹרֵי הַפְּעוּלָה. הָרִאשׁוֹן הוּא "הַגָּשַׁת בַּקָּשָׁה".

22. הוּא פּוֹתֵחַ טוֹפֶס בַּקָּשָׁה בֶּן שְׁלוֹשָׁה שְׁלָבִים, שֶׁבּוֹ אֶפְשָׁר גַּם לְהַגִּישׁ קַבָּלוֹת.

23. הַשֵּׁנִי הוּא "שְׁלִיחַת פְּנִייָה".

24. הוּא פּוֹתֵחַ טוֹפֶס פְּנִייָה לַגּוֹרֵם הַמְּטַפֵּל.

25. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר, וְהַבַּקָּשָׁה נִכְנֶסֶת לְמַצְפֵּן.

26. הַבַּקָּשָׁה מוֹפִיעָה בְּדַף הַהֲטָבָה בִּסְטָטוּס "בְּטִיפּוּל", וּמִתְעַדְכֶּנֶת שָׁם לְבַד. אֵין צוֹרֶךְ לַעֲשׂוֹת דָּבָר.

27. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת, וּבְהֶמְשֵׁךְ הוּא יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.

## scenes-invoice

1. בַּסִּרְטוֹן הַזֶּה נִרְאֶה אֵיךְ מַגִּישִׁים חֶשְׁבּוֹנִית אוֹ קַבָּלָה לְהֶחְזֵר, דֶּרֶךְ מַצְפֵּן זְכוּיּוֹת אֵיבָה. הַמַּצְפֵּן זָמִין בְּשָׁלָב זֶה לְמִשְׁפָּחוֹת שַׁכּוּלוֹת.

2. הַגָּשָׁה דֶּרֶךְ הַמַּצְפֵּן מַגִּיעָה יָשָׁר לַתִּיק שֶׁלָּכֶם, וְאֶפְשָׁר לַעֲקוֹב אַחֲרֶיהָ בְּכָל רֶגַע.

3. נִכְנָסִים מֵהָאֵזוֹר הָאִישִׁי. בַּתַּפְרִיט בּוֹחֲרִים מַצְפֵּן הַזְּכוּיּוֹת שֶׁלִּי, אַחַר כָּךְ כְּנִיסָה לְמַצְפֵּן הַזְּכוּיּוֹת, וְאָז לוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל.

4. יֵשׁ שְׁתֵּי דְּרָכִים. אִם הַהֲטָבָה נִמְצֵאת בַּהֲטָבוֹת שֶׁלִּי, מַתְחִילִים מִשָּׁם. אִם לֹא, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת. בִּשְׁנֵי הַמִּקְרִים לוֹחֲצִים עַל הַהֲטָבָה שֶׁעֲבוּרָהּ יֵשׁ לָכֶם חֶשְׁבּוֹנִית.

5. בְּדַף הַהֲטָבָה לוֹחֲצִים עַל "פְּרָטִים נוֹסָפִים".

6. וְלוֹחֲצִים עַל הַכַּפְתּוֹר הַכָּחוֹל, לְהַגָּשַׁת חֶשְׁבּוֹנִית אוֹ קַבָּלָה חֲדָשָׁה. מִכָּאן מַגִּישִׁים.

7. אִם הַהֲטָבָה לֹא מוֹפִיעָה בַּהֲטָבוֹת שֶׁלִּי, מְחַפְּשִׂים אוֹתָהּ בַּהֲטָבוֹת הַפּוֹטֶנְצִיאָלִיּוֹת וְלוֹחֲצִים עָלֶיהָ. שִׂימוּ לֵב, לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית בַּמַּצְפֵּן. תֵּדְעוּ אֶת זֶה לְפִי כַּפְתּוֹר הַפְּעוּלָה שֶׁמּוֹפִיעַ בָּהּ.

8. בָּעַמּוּד שֶׁנִּפְתָּח מוֹפִיעַ אֶחָד מִשְּׁנֵי הַכַּפְתּוֹרִים, וּלְכָל אֶחָד מֵהֶם שֵׁם מִשֶּׁלּוֹ וְתַהֲלִיךְ מִשֶּׁלּוֹ. הַכַּפְתּוֹר "הַגָּשַׁת בַּקָּשָׁה" מוֹבִיל לַמָּסָךְ שֶׁבּוֹ אֶפְשָׁר לְהַגִּישׁ חֶשְׁבּוֹנִית.

9. בִּלְחִיצָה עַל "הַגָּשַׁת בַּקָּשָׁה" נִפְתָּח טוֹפֶס הַהַגָּשָׁה.

10. הַכַּפְתּוֹר "שְׁלִיחַת פְּנִייָה" מוֹבִיל לְדַף פְּנִייָה לַגּוֹרֵם הַמְּטַפֵּל.

11. בְּהַטָּבָה שֶׁמּוֹפִיעַ בָּהּ הַכַּפְתּוֹר הַזֶּה, אִי אֶפְשָׁר בְּשָׁלָב זֶה לְהַגִּישׁ חֶשְׁבּוֹנִית יְשִׁירוֹת דֶּרֶךְ הַמַּצְפֵּן.

12. מִשְּׁתֵּי הַדְּרָכִים מַגִּיעִים לְאוֹתוֹ טוֹפֶס בֶּן שְׁלוֹשָׁה שְׁלָבִים. בַּשָּׁלָב הָרִאשׁוֹן בּוֹדְקִים אֶת פְּרָטֵי הַהֲטָבָה וְלוֹחֲצִים "הַבָּא".

13. בַּשָּׁלָב הַשֵּׁנִי מְצָרְפִים אֶת חֶשְׁבּוֹנִית הַמַּס אוֹ הַקַּבָּלָה. גּוֹרְרִים אֶת הַקּוֹבֶץ לַמִּסְגֶּרֶת, אוֹ לוֹחֲצִים "בְּחַר קוֹבֶץ".

14. אַחַר כָּךְ מְמַלְּאִים אֶת פְּרָטֵי הַחֶשְׁבּוֹנִית — מִסְפָּר, תַּאֲרִיךְ וּסְכוּם כּוֹלֵל מַס עֵרֶךְ מוּסָף. תְּקוּפָה וְכַמּוּת מְמַלְּאִים לְפִי הַצּוֹרֶךְ.

15. הַכַּפְתּוֹר "הוֹסֵף חֶשְׁבּוֹנִית" מְאַפְשֵׁר לְהוֹסִיף עוֹד חֶשְׁבּוֹנִית לְאוֹתָהּ בַּקָּשָׁה. כְּשֶׁסִּייַמְתֶּם, לוֹחֲצִים "הַבָּא".

16. קוֹרְאִים אֶת הַהַצְהָרָה וּמְסַמְּנִים אֶת הַתֵּיבָה.

17. חוֹתְמִים בַּמִּסְגֶּרֶת — בָּעַכְבָּר בַּמַּחְשֵׁב, אוֹ בָּאֶצְבַּע בַּטֶּלֶפוֹן. וְאָז שׁוֹלְחִים.

18. אַחֲרֵי הַשְּׁלִיחָה מוֹפִיעָה הוֹדָעַת אִישּׁוּר: תּוֹדָה, הַפְּנִייָה נִקְלְטָה לְהֶמְשֵׁךְ טִיפּוּל.

19. הַבַּקָּשָׁה תּוֹפִיעַ בְּדַף הַהֲטָבָה תַּחַת "בַּקָּשׁוֹת לְהֶחְזֵר", עִם הַסְּכוּם וּסְטָטוּס הַטִּיפּוּל בָּהּ. שָׁם תֵּדְעוּ אִם הַקַּבָּלָה שׁוּלְּמָה.

20. וּלְבַסּוֹף, לֹא בְּכָל הֲטָבָה פּוֹטֶנְצִיאָלִית אֶפְשָׁר לְהַגִּישׁ דֶּרֶךְ הַמַּצְפֵּן. כְּשֶׁלֹּא מוֹפִיעַ הַכַּפְתּוֹר הַגָּשַׁת בַּקָּשָׁה, הַהַגָּשָׁה נַעֲשֵׂית בְּאֶמְצָעוּת הַעֲלָאַת מִסְמָכִים בָּאֵזוֹר הָאִישִׁי. שֵׁם בּוֹחֲרִים נוֹסֵא שִׁיקּוּם, קָטֵגוֹרְיָה פְּנִיּוֹת וּמִסְמָךְ פְּנִייָה, מְצָרְפִים אֶת הַקּוֹבֶץ וְשׁוֹלְחִים.

21. בְּשָׁלָב זֶה הַמַּצְפֵּן זָמִין לְמִשְׁפָּחוֹת שַׁכּוּלוֹת, וּבְהֶמְשֵׁךְ הוּא יִיפָּתַח גַּם לְאוּכְלוּסִיּוֹת נוֹסָפוֹת. הַבִּיטּוּחַ הַלְּאוּמִּי, אֲגַף שִׁיקּוּם.
